#include <ESP32Servo.h>

Servo myServo;

int servoPin = 13;
int angle = 90;

void setup() {
  Serial.begin(115200);

  myServo.setPeriodHertz(50);
  myServo.attach(servoPin, 500, 2400);

  myServo.write(90); // start at middle
}

void loop() {
  if (Serial.available()) {
    angle = Serial.parseInt();

    if (angle >= 0 && angle <= 180) {
      myServo.write(angle);
      Serial.print("Angle: ");
      Serial.println(angle);
    }
  }
}
